cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ConsoleQueryTests LANGUAGES CXX)
    enable_testing()
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(OPENMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

add_executable(query-packet-test query-packet-test.cpp)
target_include_directories(query-packet-test PRIVATE "${OPENMP_ROOT}/Server/Components/LegacyNetwork/Query")
add_test(NAME query-packet-test COMMAND query-packet-test)
if(WIN32)
    add_executable(console-encoding-test console-encoding-test.cpp)
    target_include_directories(console-encoding-test PRIVATE "${OPENMP_ROOT}/Server/Components/Console")
    target_compile_definitions(console-encoding-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    if(MSVC)
        target_compile_options(console-encoding-test PRIVATE /utf-8)
    endif()
    add_test(NAME console-encoding-test COMMAND console-encoding-test)
endif()

option(BUILD_CONSOLE_QUERY_FIXTURES "Build isolated integration fixtures; never deploy these components" OFF)
if(BUILD_CONSOLE_QUERY_FIXTURES)
    if(NOT TARGET Server OR NOT TARGET LegacyNetwork OR NOT TARGET Console)
        message(FATAL_ERROR "Integration fixtures require a full server build with Console and LegacyNetwork.")
    endif()
    foreach(fixture NativeQueryFixture ConsoleEncodingFixture)
        if(fixture STREQUAL "NativeQueryFixture")
            set(source fixture.cpp)
        else()
            set(source console-fixture.cpp)
        endif()
        add_library(${fixture} SHARED ${source})
        target_link_libraries(${fixture} PRIVATE OMP-SDK OMP-NetCode)
        # Keep test components out of the deployable server directory.
        set_target_properties(${fixture} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    endforeach()
    if(WIN32)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME console-query-integration COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/integration.py"
            --server "$<TARGET_FILE:Server>" --network "$<TARGET_FILE:LegacyNetwork>"
            --console "$<TARGET_FILE:Console>" --query-fixture "$<TARGET_FILE:NativeQueryFixture>"
            --console-fixture "$<TARGET_FILE:ConsoleEncodingFixture>")
        set_tests_properties(console-query-integration PROPERTIES TIMEOUT 120)
    endif()
endif()
